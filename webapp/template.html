Run it on real TradingView

The TradingView widget can't load inside this page (its scripts are blocked by the artifact sandbox), so here is the EMA rule as Pine Script v5 — paste into TradingView's Pine editor → "Add to chart" to see the same entries/exits on live TradingView charts. The SMC/Elliott engines use confirmed-pivot logic that has no faithful compact Pine port; their fills are shown above instead.
//@version=5
strategy("KB swing — EMA12/21 + rules 4&5", overlay=true,
     initial_capital=1000000, default_qty_type=strategy.percent_of_equity,
     default_qty_value=100, commission_type=strategy.commission.percent, commission_value=0.1)

e12 = ta.ema(close, 12)
e21 = ta.ema(close, 21)
tp  = (high + low + close) / 3
vwap20 = math.sum(tp * volume, 20) / math.sum(volume, 20)   // rule 4 proxy
volRatio  = volume / ta.sma(volume, 20)
rng = high - low
lowerWick = rng > 0 ? (math.min(open, close) - low) / rng : 0.0

rule3 = e12 > e21
rule4 = close > vwap20
rule5 = close > open and (volRatio >= 1.5 or lowerWick >= 0.4)

var int entryBar = na
if rule3 and rule4 and rule5 and strategy.position_size == 0
    strategy.entry("L", strategy.long)
    entryBar := bar_index
exitTrend = e12 < e21
exitTime  = strategy.position_size > 0 and not na(entryBar) and bar_index - entryBar >= 126
if exitTrend or exitTime
    strategy.close("L")

plot(e12, "EMA12", color=color.new(#2a78d6, 0), linewidth=2)
plot(e21, "EMA21", color=color.new(#eb6834, 0), linewidth=2)
